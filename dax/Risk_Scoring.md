# Risk Scoring DAX MeasuresThis document contains the DAX measures used for risk scoring in the Insurance Claims Analytics project.---## Medical Risk Score```daxMedical_Risk_Score = [BMI_Risk] + [Tobacco_Risk] + [Chronic_Risk] + [Family_History_Risk]Normalized Medical Risk ScoreMedical_Risk_Normalized = DIVIDE([Medical_Risk_Score], 7, 0)Financial Risk FeaturesClaim RatioClaim_Ratio = DIVIDE([Total_Claim_Amount], [Sum_Insured], 0)Claim DelayClaim_Delay = DATEDIFF([Policy_Start_Date], [Claim_Date], DAY)Out_of_Pocket = [Deductible_Amount] + [Co_pay_Amount] + [Non_payable_Items]Income to Claim RatioIncome_to_Claim_Ratio = DIVIDE([Total_Claim_Amount], [Annual_Income_INR], 0)Normalized Features (Min-Max Normalization)Claim Ratio NormalizedClaim_Ratio_Normalized = VAR MinVal = MIN(claims[Claim_Ratio])VAR MaxVal = MAX(claims[Claim_Ratio])RETURNDIVIDE([Claim_Ratio] - MinVal, MaxVal - MinVal, 0)Cost Per Day NormalizedCost_Per_Day_Normalized = VAR MinVal = MIN(claims[Cost_Per_Day])VAR MaxVal = MAX(claims[Cost_Per_Day])RETURNDIVIDE([Cost_Per_Day] - MinVal, MaxVal - MinVal, 0)Claim Delay NormalizedClaim_Delay_Normalized = VAR MinVal = MIN(claims[Claim_Delay])VAR MaxVal = MAX(claims[Claim_Delay])RETURNDIVIDE([Claim_Delay] - MinVal, MaxVal - MinVal, 0)Out of Pocket NormalizedOut_of_Pocket_Normalized = VAR MinVal = MIN(claims[Out_of_Pocket])VAR MaxVal = MAX(claims[Out_of_Pocket])RETURNDIVIDE([Out_of_Pocket] - MinVal, MaxVal - MinVal, 0)Income to Claim Risk NormalizedIncome_to_Claim_Risk_Normalized = VAR MinVal = MIN(claims[Income_to_Claim_Ratio])VAR MaxVal = MAX(claims[Income_to_Claim_Ratio])RETURNDIVIDE([Income_to_Claim_Ratio] - MinVal, MaxVal - MinVal, 0)Overall Claim Risk ScoreOverall_Claim_Risk_Score = ([Medical_Risk_Normalized] * 0.30) +([Claim_Ratio_Normalized] * 0.25) +([Cost_Per_Day_Normalized] * 0.20) +([Claim_Delay_Normalized] * 0.10) +([Out_of_Pocket_Normalized] * 0.10) +([Income_to_Claim_Risk_Normalized] * 0.05)Risk Level ClassificationClaim_Risk_Level = SWITCH(TRUE(),    [Overall_Claim_Risk_Score] < 0.30, "Low Risk",    [Overall_Claim_Risk_Score] >= 0.30 && [Overall_Claim_Risk_Score] < 0.60, "Medium Risk",    [Overall_Claim_Risk_Score] >= 0.60 && [Overall_Claim_Risk_Score] < 0.80, "High Risk",    [Overall_Claim_Risk_Score] >= 0.80, "Potential Anomaly",    "Unclassified")last Updated: September 2025
